import vm from 'node:vm';
import fs from 'node:fs';
import {test,expect} from 'bun:test';
const source=fs.readFileSync('crates/screenpipe-core/assets/extensions/orphan-guard.ts','utf8');
const compiled=new Bun.Transpiler({loader:'ts',target:'node'}).transformSync(source).replace('export default ', 'globalThis.install = ');
// Expectations are mode/lifecycle outcomes, fixed before running any arm.
const scenarios = [
 {name:'one-shot stdin end preserves work',argv:['bun','pi','-p','synthetic prompt'],event:'end',exit:false},
 {name:'one-shot stdin close preserves work',argv:['bun','pi','-p','synthetic prompt'],event:'close',exit:false},
 {name:'RPC stdin end exits',argv:['bun','pi','--mode','rpc'],event:'end',exit:true},
 {name:'RPC stdin close exits',argv:['bun','pi','--mode','rpc'],event:'close',exit:true},
 {name:'one-shot healthy parent preserves work',argv:['bun','pi','-p','synthetic prompt'],tick:'healthy',exit:false},
 {name:'RPC healthy parent preserves work',argv:['bun','pi','--mode','rpc'],tick:'healthy',exit:false},
 {name:'one-shot changed parent exits',argv:['bun','pi','-p','synthetic prompt'],tick:'changed',exit:true},
 {name:'RPC missing parent exits',argv:['bun','pi','--mode','rpc'],tick:'missing',exit:true},
];

for(const scenario of scenarios) test(scenario.name,()=>{
  const listeners=new Map<string,Function[]>();const exits:number[]=[];let tick:Function=()=>{};
  const processPort={argv:scenario.argv,ppid:200,stdin:{on:(n:string,f:Function)=>{listeners.set(n,[...(listeners.get(n)||[]),f]);}},exit:(n:number)=>{exits.push(n);},kill:()=>{if(scenario.tick==='missing')throw Error('synthetic ESRCH');}};
  const ctx=vm.createContext({process:processPort,setInterval:(f:Function)=>{tick=f;return {unref(){}};},clearInterval:()=>{}});
  vm.runInContext(compiled+'\ninstall({});',ctx,{timeout:1000});
  if(scenario.event)for(const f of listeners.get(scenario.event)||[])f();
  if(scenario.tick){if(scenario.tick==='changed')processPort.ppid=201;tick();}
  expect(exits.length>0).toBe(scenario.exit);
});
