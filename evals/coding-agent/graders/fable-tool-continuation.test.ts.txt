import {resolve} from 'node:path';
import {expect,test} from 'bun:test';
const {createProvider}=await import(resolve(process.cwd(),'packages/ai-gateway/src/providers/index.ts'));
const checks=[];
for(const stream of [true,false]) for(const compatible of [true,false]) for(const choice of ['none','auto']) {
 const calls=[]; const body={model:'claude-fable-5',max_tokens:2048,tool_choice:choice,messages:[{role:'system',content:'Synthetic context'},{role:'user',content:'Confirm fixture'},{role:'assistant',content:null,tool_calls:[{id:'toolu_fixture',type:'function',function:{name:'confirm',arguments:'{}'}}]},{role:'tool',tool_call_id:'toolu_fixture',content:'Fixture complete'}]};const before=JSON.stringify(body);
 const provider=createProvider(body.model,{}, {apiKey:'synthetic',baseURL:'https://fixture.invalid',openAICompatible:compatible,maxRetries:0,defaultHeaders:{'X-Explabs-Tags':'{"training_allowed":false}'},fetch:async(input,init)=>{
  const req=new Request(input,init),wire=await req.json();calls.push({wire,native:!req.url.endsWith('/chat/completions'),policy:req.headers.get('X-Explabs-Tags')});
  const chat=req.url.endsWith('/chat/completions');const empty=!chat&&compatible&&choice==='none';const text=empty?'':'Fixture completed';
  if(!wire.stream)return Response.json(chat?{choices:[{message:{role:'assistant',content:text},finish_reason:'stop'}],usage:{prompt_tokens:30,completion_tokens:2}}:{id:'msg_fixture',type:'message',role:'assistant',model:body.model,content:text?[{type:'text',text}]:[],stop_reason:'end_turn',usage:{input_tokens:30,output_tokens:2}});
  if(chat)return new Response([{choices:[{delta:{content:text},finish_reason:null}]},{choices:[{delta:{},finish_reason:'stop'}]},{choices:[],usage:{prompt_tokens:30,completion_tokens:2}}].map(c=>'data: '+JSON.stringify(c)+'\n\n').join('')+'data: [DONE]\n\n',{headers:{'content-type':'text/event-stream'}});
  const events=[['message_start',{type:'message_start',message:{id:'msg_fixture',type:'message',role:'assistant',model:body.model,content:[],usage:{input_tokens:30,output_tokens:0}}}],...(text?[['content_block_start',{type:'content_block_start',index:0,content_block:{type:'text',text:''}}],['content_block_delta',{type:'content_block_delta',index:0,delta:{type:'text_delta',text}}],['content_block_stop',{type:'content_block_stop',index:0}]]:[]),['message_delta',{type:'message_delta',delta:{stop_reason:'end_turn'},usage:{output_tokens:2}}],['message_stop',{type:'message_stop'}]];
  return new Response(events.map(([event,data])=>'event: '+event+'\ndata: '+JSON.stringify(data)+'\n\n').join(''),{headers:{'content-type':'text/event-stream'}});
 }});
 const result=stream?await new Response(await provider.createStreamingCompletion(body)).text():JSON.stringify(await (await provider.createCompletion(body)).json());
 checks.push({stream,compatible,choice,completed:result.includes('Fixture completed'),single_dispatch:calls.length===1,budget:calls.every(c=>c.wire.max_tokens===2048),policy:calls.every(c=>c.policy==='{"training_allowed":false}'),input_unchanged:JSON.stringify(body)===before,native_mode:compatible||calls.every(c=>c.native),usage:result.includes('"prompt_tokens":30')&&result.includes('"completion_tokens":2')});
}
for(const c of checks)for(const field of ['completed','single_dispatch','budget','policy','input_unchanged','native_mode','usage'])test(`${c.stream?'stream':'JSON'} compatible=${c.compatible} tool_choice=${c.choice} ${field}`,()=>expect(c[field]).toBe(true));
